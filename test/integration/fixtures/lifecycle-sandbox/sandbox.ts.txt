import { execFile } from "node:child_process";
import { access, cp, mkdir, mkdtemp, readFile, rm, writeFile } from "node:fs/promises";
import { dirname, isAbsolute, join, resolve } from "node:path";
import { Effect } from "effect";
import {
  CustomSandboxMaterializationError,
  defineSandboxCase,
  type CommandOptions,
  type SandboxTargetPlatform,
} from "niceeval/sandbox";
import { harness, projectRoot, record } from "./journal.ts";

function hostPlatform(): SandboxTargetPlatform {
  if (process.platform === "darwin") return { _tag: "Darwin", os: "darwin", arch: process.arch };
  const report = process.report.getReport() as { header?: { glibcVersionRuntime?: string } };
  return { _tag: "Linux", os: "linux", arch: process.arch, libc: report.header?.glibcVersionRuntime === undefined ? "musl" : "gnu" };
}

/** Companion resources are real directories next to the project: the primary workspace plus one directory per companion. */
export const COMPANIONS = 2;
let counter = 0;

async function materialize() {
  const index = counter++;
  if (harness().failCreates.has(index)) {
    record("create-failed", { index });
    throw new Error(`provider refused materialization ${index}`);
  }
  const root = await mkdtemp(join(projectRoot, `resource-group-${index}-`));
  const workdir = join(root, "workspace");
  const privateTmp = join(root, "tmp");
  const companions = Array.from({ length: COMPANIONS }, (_, number) => join(root, `companion-${number}`));
  await Promise.all([workdir, privateTmp, ...companions].map((path) => mkdir(path)));
  const sandboxId = `lifecycle-${root.split("/").at(-1)}`;
  record("create", { sandboxId, root });
  const map = (value: string) => value.replaceAll("/tmp/.niceeval-ledger", join(privateTmp, ".niceeval-ledger"));
  const at = (path: string) => (isAbsolute(map(path)) ? map(path) : resolve(workdir, map(path)));
  const pending = new Map<string, () => void>();
  const probeName = "niceeval-test-background";
  async function backgroundProbe(args: readonly string[], options: CommandOptions) {
    const [action, label = "", status = "0"] = args;
    const result = (stdout: string, exitCode = Number(status)) => ({ stdout, stderr: "", exitCode });
    if (action === "missing") {
      throw Object.assign(new Error("Test executable unavailable"), { code: "ENOENT", syscall: "spawn" });
    }
    if (action === "active") return result([...pending.keys()].sort().join(","));
    if (action === "release") {
      pending.get(label)?.();
      return result(label);
    }
    if (action === "exit") return result(label);
    const finished = Promise.withResolvers<ReturnType<typeof result>>();
    const signal = options.signal;
    signal?.throwIfAborted();
    const abort = () => {
      record("background-terminated", { label, sandboxId });
      finished.reject(signal?.reason);
    };
    pending.set(label, () => finished.resolve(result(label)));
    signal?.addEventListener("abort", abort, { once: true });
    const timer = options.timeoutMs === undefined ? undefined : setTimeout(() => {
      finished.reject(new DOMException("Test command timed out", "TimeoutError"));
    }, options.timeoutMs);
    try {
      void options.onStdout?.(`running:${label}`);
      return await finished.promise;
    } finally {
      if (timer !== undefined) clearTimeout(timer);
      pending.delete(label);
      signal?.removeEventListener("abort", abort);
    }
  }
  const localCommands = new Set<Promise<unknown>>();
  let stopping = false;
  const runLocal = (command: string, args: readonly string[] = [], options: CommandOptions = {}) => {
    if (stopping) return Promise.reject(new Error("Lifecycle Sandbox is stopping"));
    const result = new Promise<{ stdout: string; stderr: string; exitCode: number }>((done) => {
      record("command-start", { sandboxId, command, args });
      execFile(
        map(command),
        args.map(map),
        {
          cwd: at(options.cwd ?? workdir),
          env: {
            PATH: process.env.PATH,
            HOME: root,
            TMPDIR: privateTmp,
            LANG: "C.UTF-8",
            ...Object.fromEntries(
              Object.entries(options.env ?? {}).flatMap(([key, value]) => (value === undefined ? [] : [[key, map(value)]])),
            ),
          },
          maxBuffer: 16 * 1024 * 1024,
        },
        (error, stdout, stderr) => {
          record("command-end", { sandboxId, command, args });
          done({ stdout, stderr, exitCode: error ? (typeof error.code === "number" ? error.code : 1) : 0 });
        },
      );
    });
    localCommands.add(result);
    void result.then(() => localCommands.delete(result), () => localCommands.delete(result));
    return result;
  };
  const run = (command: string, args: readonly string[] = [], options: CommandOptions = {}) =>
    command === probeName ? backgroundProbe(args, options) : runLocal(command, args, options);
  const writeBytes = async (path: string, bytes: Uint8Array) => {
    await mkdir(dirname(at(path)), { recursive: true });
    await writeFile(at(path), bytes);
  };
  const sandbox = {
    sandboxId,
    workdir,
    otlpHost: "localhost",
    runCommand: run,
    runShell: (script: string, options?: CommandOptions) =>
      script.startsWith(`${probeName} `) ? run(probeName, script.split(" ").slice(1), options) : run("bash", ["-c", script], options),
    readText: (path: string) => readFile(at(path), "utf8"),
    writeText: (path: string, text: string) => writeBytes(path, Buffer.from(text, "utf8")),
    readBytes: (path: string) => readFile(at(path)),
    writeBytes,
    pathExists: (path: string) => access(at(path)).then(() => true, () => false),
    uploadFile: async (source: string, target: string) => writeBytes(target, await readFile(source)),
    uploadDirectory: (source: string, target?: string) => cp(source, at(target ?? "."), { recursive: true }),
    downloadFile: async (source: string, target: string | URL) => {
      await mkdir(dirname(String(target)), { recursive: true });
      await writeFile(target, await readFile(at(source)));
    },
    downloadDirectory: (source: string, target: string) => cp(at(source), target, { recursive: true }),
    stop: async () => undefined,
  };
  let stops = 0;
  const group = {
    primary: { sandboxId, provider: "lifecycle-host" },
    resources: { root, companions },
    // Settling the group must release the primary workspace and every companion exactly once.
    stop: async () => {
      stops += 1;
      stopping = true;
      record("stop", { sandboxId, stops, pendingCommands: [...pending.keys()] });
      try {
        // These short host commands outlive an interrupted Effect Promise. A real
        // provider destroys their Sandbox; this directory-backed fixture must
        // join them before deleting the tree, or ledger setup can recreate it.
        await Promise.allSettled(localCommands);
        await rm(root, { recursive: true, force: true });
        record("stop-completed", { sandboxId });
      } catch (error) {
        record("stop-failed", { sandboxId, error: String(error) });
        throw error;
      }
    },
  };
  return { sandbox, group, services: { _tag: "None" as const }, facts: {} };
}

export default defineSandboxCase({
  identity: { kind: "lifecycle-host", companions: COMPANIONS },
  targetPlatform: hostPlatform(),
  services: { _tag: "Unsupported" },
  materialize: () =>
    Effect.tryPromise({
      try: materialize,
      catch: (cause) =>
        new CustomSandboxMaterializationError({
          code: "lifecycle-create-failed",
          message: String(cause),
          cause: cause instanceof Error ? cause : new Error(String(cause)),
        }),
    }),
});
