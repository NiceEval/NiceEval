import { defineEval } from "niceeval";
import { loadJson } from "niceeval/loaders";

const turns = await loadJson(new URL("./CASE_FILE.json", import.meta.url), (value) => {
  if (!Array.isArray(value) || !value.every((paths) => Array.isArray(paths) && paths.every((path) => typeof path === "string"))) {
    throw new TypeError("Expected an array of path arrays");
  }
  return value as string[][];
});

export default defineEval({
  async test(t) {
    for (const [index, paths] of turns.entries()) {
      const turn = await t.send(JSON.stringify({ turn: index + 1, paths }));
      if (turns.length === 4) {
        turn.sandbox.noChanges().label(`turn ${index + 1} has no changes`);
        continue;
      }
      turn.sandbox.changedPaths(paths).label(`turn ${index + 1} matches`);
      turn.sandbox.changedPaths(
        paths.includes("a.txt") ? paths.filter((path) => path !== "a.txt") : [...paths, "a.txt"],
      ).label(`turn ${index + 1} mismatches`);
    }
    t.sandbox.changedPaths([...new Set(turns.flat())]).label("Attempt union");
  },
});
